```javascript
// router/permission.js - 路由守卫
router.beforeEach((to, from, next) => {
  const token = getToken()
  
  // 需要登录
  if (to.meta.requireAuth && !token) {
    return next('/login')
  }
  
  // 权限检查
  if (to.meta.permission && !hasPermission(to.meta.permission)) {
    return next('/403')
  }
  
  next()
})
```

```javascript
// stores/user.js
import { defineStore } from 'pinia'

export const useUserStore = defineStore('user', {
  state: () => ({
    token: getToken(),
    userInfo: null,
    permissions: [],
    roles: []
  }),
  
  getters: {
    isAdmin: (state) => state.roles.includes('admin'),
    hasPermission: (state) => (permission) => {
      return state.permissions.includes(permission)
    }
  },
  
  actions: {
    async login(username, password) {
      const res = await loginAPI({ username, password })
      this.token = res.token
      setToken(res.token)
    },
    
    async getUserInfo() {
      const res = await getUserInfoAPI()
      this.userInfo = res.user
      this.permissions = res.permissions
      this.roles = res.roles
    },
    
    logout() {
      this.token = null
      this.userInfo = null
      this.permissions = []
      this.roles = []
      removeToken()
    }
  }
})
```

```javascript
// controllers/customerController.js
const customerService = require('../services/customerService')
const { successResponse, errorResponse } = require('../utils/response')

class CustomerController {
  // 获取客户列表
  async getList(req, res, next) {
    try {
      const { page = 1, pageSize = 20, keyword, stage } = req.query
      const userId = req.user.id
      const userRole = req.user.role
      
      const result = await customerService.getList({
        page,
        pageSize,
        keyword,
        stage,
        userId,
        userRole
      })
      
      return successResponse(res, result)
    } catch (error) {
      next(error)
    }
  }
  
  // 获取客户详情
  async getDetail(req, res, next) {
    try {
      const { id } = req.params
      const userId = req.user.id
      
      const customer = await customerService.getDetail(id, userId)
      
      if (!customer) {
        return errorResponse(res, '客户不存在', 404)
      }
      
      return successResponse(res, customer)
    } catch (error) {
      next(error)
    }
  }
  
  // 添加客户
  async create(req, res, next) {
    try {
      const data = req.body
      const userId = req.user.id
      
      const customer = await customerService.create({
        ...data,
        ownerId: userId
      })
      
      return successResponse(res, customer, '添加成功', 201)
    } catch (error) {
      next(error)
    }
  }
  
  // 更新客户
  async update(req, res, next) {
    try {
      const { id } = req.params
      const data = req.body
      const userId = req.user.id
      
      const customer = await customerService.update(id, data, userId)
      
      return successResponse(res, customer, '更新成功')
    } catch (error) {
      next(error)
    }
  }
  
  // 删除客户
  async delete(req, res, next) {
    try {
      const { id } = req.params
      const userId = req.user.id
      
      await customerService.delete(id, userId)
      
      return successResponse(res, null, '删除成功')
    } catch (error) {
      next(error)
    }
  }
  
  // 转移客户
  async transfer(req, res, next) {
    try {
      const { id } = req.params
      const { targetUserId, reason } = req.body
      const userId = req.user.id
      
      await customerService.transfer(id, targetUserId, reason, userId)
      
      return successResponse(res, null, '转移成功')
    } catch (error) {
      next(error)
    }
  }
}

module.exports = new CustomerController()
```

#### 3.3 服务层模块

```javascript
// services/customerService.js
const { Customer, Followup, User } = require('../models')
const { Op } = require('sequelize')

class CustomerService {
  // 获取客户列表
  async getList({ page, pageSize, keyword, stage, userId, userRole }) {
    const where = {}
    
    // 数据权限过滤
    if (userRole === 'salesperson') {
      where.ownerId = userId // 销售只能看自己的客户
    } else if (userRole === 'manager') {
      // 经理能看团队客户（需要查询团队成员）
      const teamUserIds = await this.getTeamUserIds(userId)
      where.ownerId = { [Op.in]: teamUserIds }
    }
    // admin和executive能看所有客户，不加过滤条件
    
    // 关键词搜索
    if (keyword) {
      where[Op.or] = [
        { name: { [Op.like]: `%${keyword}%` } },
        { company: { [Op.like]: `%${keyword}%` } },
        { phone: { [Op.like]: `%${keyword}%` } }
      ]
    }
    
    // 销售阶段筛选
    if (stage) {
      where.stage = stage
    }
    
    const { count, rows } = await Customer.findAndCountAll({
      where,
      include: [
        {
          model: User,
          as: 'owner',
          attributes: ['id', 'name']
        }
      ],
      limit: parseInt(pageSize),
      offset: (parseInt(page) - 1) * parseInt(pageSize),
      order: [['lastFollowupAt', 'DESC']]
    })
    
    return {
      list: rows,
      total: count,
      page: parseInt(page),
      pageSize: parseInt(pageSize)
    }
  }
  
  // 获取客户详情
  async getDetail(id, userId) {
    const customer = await Customer.findByPk(id, {
      include: [
        {
          model: User,
          as: 'owner',
          attributes: ['id', 'name']
        },
        {
          model: Followup,
          as: 'followups',
          order: [['createdAt', 'DESC']],
          limit: 20
        }
      ]
    })
    
    // 权限检查：是否有权限查看该客户
    // （在Controller或Middleware中实现更合适）
    
    return customer
  }
  
  // 添加客户
  async create(data) {
    // 检查手机号是否重复
    const exists = await Customer.findOne({
      where: { phone: data.phone }
    })
    
    if (exists) {
      throw new Error('该客户已存在')
    }
    
    // 生成客户编号
    data.customerNo = await this.generateCustomerNo()
    
    const customer = await Customer.create(data)
    
    return customer
  }
  
  // 更新客户
  async update(id, data, userId) {
    const customer = await Customer.findByPk(id)
    
    if (!customer) {
      throw new Error('客户不存在')
    }
    
    // 权限检查：是否有权限修改该客户
    
    await customer.update(data)
    
    return customer
  }
  
  // 删除客户（软删除）
  async delete(id, userId) {
    const customer = await Customer.findByPk(id)
    
    if (!customer) {
      throw new Error('客户不存在')
    }
    
    // 权限检查
    
    await customer.destroy() // Sequelize默认软删除（如果配置了paranoid: true）
  }
  
  // 转移客户
  async transfer(id, targetUserId, reason, userId) {
    const customer = await Customer.findByPk(id)
    
    if (!customer) {
      throw new Error('客户不存在')
    }
    
    // 开启事务
    const transaction = await Customer.sequelize.transaction()
    
    try {
      // 更新客户负责人
      await customer.update({ ownerId: targetUserId }, { transaction })
      
      // 记录转移日志
      await TransferLog.create({
        customerId: id,
        fromUserId: userId,
        toUserId: targetUserId,
        reason
      }, { transaction })
      
      await transaction.commit()
    } catch (error) {
      await transaction.rollback()
      throw error
    }
  }
  
  // 生成客户编号
  async generateCustomerNo() {
    const date = new Date()
    const dateStr = date.toISOString().slice(0, 10).replace(/-/g, '')
    const count = await Customer.count({
      where: {
        customerNo: { [Op.like]: `C${dateStr}%` }
      }
    })
    const serial = String(count + 1).padStart(4, '0')
    return `C${dateStr}${serial}`
  }
  
  // 获取团队成员ID
  async getTeamUserIds(managerId) {
    const users = await User.findAll({
      where: { managerId },
      attributes: ['id']
    })
    return users.map(u => u.id)
  }
}

module.exports = new CustomerService()
```

#### 3.4 中间件模块

```javascript
// middlewares/auth.js - 身份验证
const jwt = require('jsonwebtoken')
const { errorResponse } = require('../utils/response')

exports.auth = (req, res, next) => {
  const token = req.headers.authorization?.replace('Bearer ', '')
  
  if (!token) {
    return errorResponse(res, '未登录', 401)
  }
  
  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET)
    req.user = decoded
    next()
  } catch (error) {
    return errorResponse(res, 'Token无效或已过期', 401)
  }
}

// middlewares/permission.js - 权限检查
exports.permission = (requiredPermission) => {
  return async (req, res, next) => {
    const userPermissions = req.user.permissions || []
    
    if (!userPermissions.includes(requiredPermission)) {
      return errorResponse(res, '无权限操作', 403)
    }
    
    next()
  }
}

// middlewares/errorHandler.js - 统一错误处理
exports.errorHandler = (err, req, res, next) => {
  const logger = require('../utils/logger')
  
  // 记录错误日志
  logger.error({
    message: err.message,
    stack: err.stack,
    url: req.url,
    method: req.method,
    ip: req.ip,
    user: req.user?.id
  })
  
  // 返回错误响应
  const status = err.status || 500
  const message = process.env.NODE_ENV === 'production' 
    ? '服务器内部错误' 
    : err.message
  
  return errorResponse(res, message, status)
}
```

## 🔌 接口设计规范

### 1. RESTful API规范

#### 资源命名
- 使用名词复数形式：`/api/customers`（而非`/api/getCustomers`）
- 使用小写字母，单词间用连字符：`/api/customer-tags`
- 资源嵌套不超过2层：`/api/customers/:id/followups`

#### HTTP方法
| 方法 | 用途 | 示例 |
|------|------|------|
| GET | 查询 | `GET /api/customers` - 获取列表 |
| | | `GET /api/customers/:id` - 获取详情 |
| POST | 创建 | `POST /api/customers` - 创建客户 |
| PUT | 完整更新 | `PUT /api/customers/:id` - 更新客户（所有字段）|
| PATCH | 部分更新 | `PATCH /api/customers/:id` - 更新客户（部分字段）|
| DELETE | 删除 | `DELETE /api/customers/:id` - 删除客户 |

## 🔐 安全设计

### 1. 认证机制
- **JWT Token**：有效期2小时，存储用户ID、角色、权限
- **RefreshToken**：有效期7天，用于刷新AccessToken
- **密码加密**：BCrypt加密，盐值10轮

### 2. 权限控制
- **功能权限**：基于角色的权限控制（RBAC）
- **数据权限**：基于用户角色和归属关系过滤数据

### 3. 接口安全
- **HTTPS**：生产环境强制HTTPS
- **CORS**：配置允许的域名白名单
- **限流**：单IP每分钟最多60次请求
- **SQL注入防护**：使用ORM参数化查询
- **XSS防护**：前端输入过滤，后端输出转义
