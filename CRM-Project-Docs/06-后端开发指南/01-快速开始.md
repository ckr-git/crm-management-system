# 后端开发快速开始

> 📌 本文档帮助你快速搭建CRM后端开发环境并开始开发

---

## 🚀 环境准备

### 1. 安装Node.js
下载并安装 Node.js 18 LTS版本

**验证安装**：
```bash
node -v  # 应显示 v18.x.x
npm -v   # 应显示 9.x.x
```

---

### 2. 安装MySQL 8.0
下载并安装MySQL 8.0

**创建数据库**：
```sql
CREATE DATABASE crm_system CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
```

---

### 3. 安装Redis
下载并安装Redis 7.2（可选，用于缓存和Session）

---

### 4. 业务逻辑层

**创建 src/services/customerService.js**：
```javascript
const { Customer, User, Followup } = require('../models')
const { Op } = require('sequelize')

class CustomerService {
  async getList({ page = 1, pageSize = 20, keyword, stage, userId, userRole }) {
    const where = {}
    
    // 数据权限过滤
    if (userRole === 'salesperson') {
      where.owner_id = userId
    }
    
    // 关键词搜索
    if (keyword) {
      where[Op.or] = [
        { name: { [Op.like]: `%${keyword}%` } },
        { company: { [Op.like]: `%${keyword}%` } },
        { phone: { [Op.like]: `%${keyword}%` } }
      ]
    }
    
    // 阶段筛选
    if (stage) {
      where.stage = stage
    }
    
    const { count, rows } = await Customer.findAndCountAll({
      where,
      include: [
        {
          model: User,
          as: 'owner',
          attributes: ['id', 'name']
        }
      ],
      limit: parseInt(pageSize),
      offset: (parseInt(page) - 1) * parseInt(pageSize),
      order: [['last_followup_at', 'DESC']]
    })
    
    return {
      list: rows,
      total: count,
      page: parseInt(page),
      pageSize: parseInt(pageSize)
    }
  }
  
  async getDetail(id) {
    const customer = await Customer.findByPk(id, {
      include: [
        {
          model: User,
          as: 'owner',
          attributes: ['id', 'name', 'phone']
        },
        {
          model: Followup,
          as: 'followups',
          limit: 20,
          order: [['followup_at', 'DESC']],
          include: [
            {
              model: User,
              as: 'user',
              attributes: ['id', 'name']
            }
          ]
        }
      ]
    })
    
    return customer
  }
  
  async create(data) {
    // 检查手机号是否重复
    const exists = await Customer.findOne({
      where: { phone: data.phone }
    })
    
    if (exists) {
      throw new Error('该手机号已存在')
    }
    
    // 生成客户编号
    data.customer_no = await this.generateCustomerNo()
    
    const customer = await Customer.create(data)
    
    return customer
  }
  
  async update(id, data) {
    const customer = await Customer.findByPk(id)
    
    if (!customer) {
      throw new Error('客户不存在')
    }
    
    await customer.update(data)
    
    return customer
  }
  
  async delete(id) {
    const customer = await Customer.findByPk(id)
    
    if (!customer) {
      throw new Error('客户不存在')
    }
    
    await customer.destroy()
  }
  
  async generateCustomerNo() {
    const date = new Date()
    const dateStr = date.toISOString().slice(0, 10).replace(/-/g, '')
    const count = await Customer.count({
      where: {
        customer_no: { [Op.like]: `C${dateStr}%` }
      }
    })
    const serial = String(count + 1).padStart(4, '0')
    return `C${dateStr}${serial}`
  }
}

module.exports = new CustomerService()
```

### 5. 控制器层

**创建 src/controllers/customerController.js**：
```javascript
const customerService = require('../services/customerService')
const { successResponse, errorResponse } = require('../utils/response')

class CustomerController {
  async getList(req, res, next) {
    try {
      const { page, pageSize, keyword, stage } = req.query
      const userId = req.user.id
      const userRole = req.user.role
      
      const result = await customerService.getList({
        page,
        pageSize,
        keyword,
        stage,
        userId,
        userRole
      })
      
      return successResponse(res, result)
    } catch (error) {
      next(error)
    }
  }
  
  async getDetail(req, res, next) {
    try {
      const { id } = req.params
      const customer = await customerService.getDetail(id)
      
      if (!customer) {
        return errorResponse(res, '客户不存在', 404)
      }
      
      return successResponse(res, customer)
    } catch (error) {
      next(error)
    }
  }
  
  async create(req, res, next) {
    try {
      const data = req.body
      const userId = req.user.id
      
      const customer = await customerService.create({
        ...data,
        owner_id: userId,
        creator_id: userId
      })
      
      return successResponse(res, customer, '创建成功', 201)
    } catch (error) {
      next(error)
    }
  }
  
  async update(req, res, next) {
    try {
      const { id } = req.params
      const data = req.body
      
      const customer = await customerService.update(id, data)
      
      return successResponse(res, customer, '更新成功')
    } catch (error) {
      next(error)
    }
  }
  
  async delete(req, res, next) {
    try {
      const { id } = req.params
      await customerService.delete(id)
      
      return successResponse(res, null, '删除成功')
    } catch (error) {
      next(error)
    }
  }
}

module.exports = new CustomerController()
```

### 7. 中间件

**创建 src/middlewares/auth.js**：
```javascript
const jwt = require('jsonwebtoken')
const { errorResponse } = require('../utils/response')

exports.auth = (req, res, next) => {
  const token = req.headers.authorization?.replace('Bearer ', '')
  
  if (!token) {
    return errorResponse(res, '未登录', 401)
  }
  
  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET)
    req.user = decoded
    next()
  } catch (error) {
    return errorResponse(res, 'Token无效或已过期', 401)
  }
}
```

**创建 src/middlewares/errorHandler.js**：
```javascript
const { errorResponse } = require('../utils/response')

const errorHandler = (err, req, res, next) => {
  console.error('Error:', err)
  
  const status = err.status || 500
  const message = process.env.NODE_ENV === 'production' 
    ? '服务器内部错误' 
    : err.message
  
  return errorResponse(res, message, status)
}

module.exports = errorHandler
```
